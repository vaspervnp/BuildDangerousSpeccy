; Bambis, the owner's nephew (Sunny Vale east wing, room 1: paces the ground floor,
; columns 18-30). Walking into him costs a selfie; then he rests a few seconds before the
; next. A jump clears him.

poke calls_enabled 0              ; no phone calls in this script (tests/phone.txt has them)
poke workers_enabled 0            ; no workers (tests/workers.txt)
pokew boot_campaign campaign_single
run 5
run 1 fire                      ; title: start
run 3
run 1 fire                      ; skip the outdoor intro
until game_state 0 -
expect npc_on 0                 ; not in the lobby

goto 1 0 20 right               ; into the east wing, ground floor
expect npc_on 1
expect npc_y 20
shot nephew_room
until pl_state 4 right 300      ; keep walking: he gets his selfie
expect npc_done 200               ; NEPHEW_REST
expect npc_pose 1
run 75                          ; the caption, then the first flash
expect selfie_step 2
shot nephew_flash
until pl_state 0 - 400          ; flash, flash, "like and subscribe", done
expect npc_pose 0
expect selfie_index 1
shot nephew_after
goto 1 26 20 right              ; straight past him while he rests
expect pl_state 0

goto 0 30 20 left               ; out and back in: he's back for more
goto 1 0 20 right
expect npc_done 0
until pl_state 4 right 300
until pl_state 0 - 400
expect selfie_index 2

goto 1 26 20 right              ; past him while he rests, to the right of his beat
until npc_done 0 - 300          ; rested: he wants another
until npc_dir 0 - 200           ; coming this way...
until npc_x 20 - 200
run 1 up+left                   ; ...hop over him
run 35 left
expect pl_state 0
expect selfie_index 2
shot nephew_jumped
goto 1 4 20 left                ; and away: he can't keep up
expect selfie_index 2
