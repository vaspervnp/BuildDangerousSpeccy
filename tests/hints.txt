; First-trap hints (hints.asm): the first three traps Thanos comes within reach of open a
; window telling him to press SPACE; the third adds that he's on his own from now on.
; Sunny Vale's route: lobby sockets at 8,20 and 27,13, the east wing's pipe at 14,13,
; then the ferret grille at 16,21 - no window for that one.
; With both lobby traps gone off, a window says to sign the room off with M, until he does.

poke calls_enabled 0              ; no phone calls in this script (tests/phone.txt has them)
poke nephew_enabled 0             ; no selfies (tests/nephew.txt)
poke workers_enabled 0            ; no workers (tests/workers.txt)
pokew boot_campaign campaign_single
run 5
run 1 fire                      ; title: start
run 3
run 1 fire                      ; skip the outdoor intro
until game_state 0 -
run 1
expect hint_count 0
expect hint_row 0

; --- the first trap: a window at the top (he's on the ground floor)
goto 0 6 20 right
run 2
expect hint_count 1
expect hint_row 2
shot hint_first
goto 0 3 20 left                ; walk away: it closes
expect hint_row 0
goto 0 6 20 right              ; back again: it opens, still the first
run 2
expect hint_row 2
expect hint_count 1
run 1 fire                      ; SPACE: the trap goes off, the window goes
run 2
expect hint_row 0
until pl_state 0 - 300
run 2
expect hint_row 0               ; disarmed: nothing to hint

; --- the second, in the lobby's 1st floor
goto 0 12 20 right
goto 0 19 13 up
goto 0 25 13 right
run 2
expect hint_count 2
expect hint_row 2
run 1 fire
until pl_state 0 - 300

; --- the lobby is clear: sign it off
run 2
expect hint_kind 3
expect hint_row 2
shot hint_mark
expect hint_marked 0
run 1 mark
run 2
expect hint_row 0
expect hint_marked 1

; --- the third, in the east wing: the long version
goto 1 12 13 right
run 2
expect hint_count 3
expect hint_row 2
shot hint_third
run 1 fire
until pl_state 0 - 300
run 2
expect hint_row 0               ; the east wing is clear, but he knows about M now

; --- the fourth: on his own
goto 1 3 13 left
goto 1 10 20 down
goto 1 14 20 right
run 2
expect hint_count 3
expect hint_row 0
shot hint_none
